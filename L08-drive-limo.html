8.1The idea in 10 minutes

⏱️ Simulation time

Gazebo has its own clock, published on /clock. In a slow VM, 1 simulated second can take 2 real seconds. Add --ros-args -p use_sim_time:=true when you run your nodes, so their timers follow Gazebo's clock and not the wall clock.

2Write drive_square.py

Plan: drive straight for side / speed seconds, then turn for (π/2) / turn_speed seconds, 4 times. The timer runs at 10 Hz, so we count ticks. This is open loop: we never check where the robot really is.

📄 limo_lab/limo_lab/drive_square.py
#!/usr/bin/env python3
import math
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist


class DriveSquare(Node):
    def __init__(self):
        super().__init__("drive_square")
        self.declare_parameter("side", 0.5)         # metres
        self.declare_parameter("speed", 0.2)        # m/s
        self.declare_parameter("turn_speed", 0.5)   # rad/s
        self.side_ = self.get_parameter("side").value
        self.speed_ = self.get_parameter("speed").value
        self.turn_speed_ = self.get_parameter("turn_speed").value

        self.dt_ = 0.1                                               # 10 Hz
        self.forward_ticks_ = round(self.side_ / self.speed_ / self.dt_)
        self.turn_ticks_ = round((math.pi / 2) / self.turn_speed_ / self.dt_)

        self.phase_ = "forward"
        self.ticks_ = 0
        self.sides_done_ = 0
        self.cmd_pub_ = self.create_publisher(Twist, "cmd_vel", 10)
        self.timer_ = self.create_timer(self.dt_, self.loop)
        self.get_logger().info(f"Square: {self.forward_ticks_} ticks forward, {self.turn_ticks_} ticks turning")

    def loop(self):
        msg = Twist()                              # all zeros = stop
        if self.sides_done_ >= 4:
            self.cmd_pub_.publish(msg)
            self.get_logger().info("Square finished")
            self.timer_.cancel()
            return

        if self.phase_ == "forward":
            msg.linear.x = self.speed_
            self.ticks_ += 1
            if self.ticks_ >= self.forward_ticks_:
                self.phase_, self.ticks_ = "turn", 0
        else:
            msg.angular.z = self.turn_speed_
            self.ticks_ += 1
            if self.ticks_ >= self.turn_ticks_:
                self.phase_, self.ticks_ = "forward", 0
                self.sides_done_ += 1
                self.get_logger().info(f"Side {self.sides_done_} done")

        self.cmd_pub_.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(DriveSquare())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

L4Lab 4 — Read odometry

LAB 4 · 25 min

Where am I, and which way am I facing?

🧭 Quaternion → yaw in one formula

3D orientation is stored as a quaternion (x, y, z, w) to avoid problems with angles. A ground robot only needs the heading around the vertical axis, yaw (the turtle's theta):

yaw = atan2( 2(w·z + x·y), 1 − 2(y² + z²) )

Facing +x → yaw 0. Turned left 90° → yaw +1.57. Facing backwards → ±3.14.

1Write odom_reader.py
📄 limo_lab/limo_lab/odom_reader.py
#!/usr/bin/env python3
import math
import rclpy
from rclpy.node import Node
from nav_msgs.msg import Odometry


def yaw_from_quaternion(q):
    return math.atan2(2.0 * (q.w * q.z + q.x * q.y), 1.0 - 2.0 * (q.y * q.y + q.z * q.z))


class OdomReader(Node):
    def __init__(self):
        super().__init__("odom_reader")
        self.x_ = self.y_ = self.yaw_ = None
        self.create_subscription(Odometry, "odom", self.callback_odom, 10)
        self.create_timer(1.0, self.report)          # print once per second, not 50x

    def callback_odom(self, msg):
        self.x_ = msg.pose.pose.position.x
        self.y_ = msg.pose.pose.position.y
        self.yaw_ = yaw_from_quaternion(msg.pose.pose.orientation)

    def report(self):
        if self.x_ is not None:
            self.get_logger().info(
                f"x={self.x_:.2f} m  y={self.y_:.2f} m  yaw={math.degrees(self.yaw_):.0f} deg")


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(OdomReader())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

L5Lab 5 — Go-to-goal on the LIMO (closed loop)

LAB 5 · 40 min

Your Lecture 7 controller, now in metres

It is the same idea as the turtle controller, with three safety upgrades for a real robot: speed limits, turn first if the heading is badly off, and a tighter stop distance (5 cm).

1Write go_to_goal.py
📄 limo_lab/limo_lab/go_to_goal.py
#!/usr/bin/env python3
import math
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist
from nav_msgs.msg import Odometry


def yaw_from_quaternion(q):
    return math.atan2(2.0 * (q.w * q.z + q.x * q.y), 1.0 - 2.0 * (q.y * q.y + q.z * q.z))


def clamp(value, limit):
    return max(-limit, min(limit, value))


class GoToGoal(Node):
    def __init__(self):
        super().__init__("go_to_goal")
        self.declare_parameter("goal_x", 1.0)
        self.declare_parameter("goal_y", 0.5)
        self.goal_x_ = self.get_parameter("goal_x").value
        self.goal_y_ = self.get_parameter("goal_y").value

        self.k_lin_, self.k_ang_ = 0.5, 1.5          # P gains
        self.max_lin_, self.max_ang_ = 0.3, 1.0      # safety limits (m/s, rad/s)
        self.tolerance_ = 0.05                       # stop within 5 cm
        self.pose_ = None                            # (x, y, yaw)
        self.reached_ = False

        self.cmd_pub_ = self.create_publisher(Twist, "cmd_vel", 10)
        self.create_subscription(Odometry, "odom", self.callback_odom, 10)
        self.create_timer(0.05, self.control_loop)   # 20 Hz
        self.get_logger().info(f"Going to ({self.goal_x_}, {self.goal_y_})")

    def callback_odom(self, msg):
        p = msg.pose.pose
        self.pose_ = (p.position.x, p.position.y, yaw_from_quaternion(p.orientation))

    def control_loop(self):
        if self.pose_ is None or self.reached_:
            return
        x, y, yaw = self.pose_
        dist_x = self.goal_x_ - x
        dist_y = self.goal_y_ - y
        distance = math.sqrt(dist_x * dist_x + dist_y * dist_y)

        msg = Twist()
        if distance > self.tolerance_:
            diff = math.atan2(dist_y, dist_x) - yaw
            diff = math.atan2(math.sin(diff), math.cos(diff))       # wrap to [-pi, pi]
            msg.angular.z = clamp(self.k_ang_ * diff, self.max_ang_)
            if abs(diff) < 0.4:                                     # roughly facing the goal?
                msg.linear.x = clamp(self.k_lin_ * distance, self.max_lin_)
        else:
            self.reached_ = True
            self.get_logger().info(f"Goal reached! final error {distance * 100:.1f} cm")
        self.cmd_pub_.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(GoToGoal())
    rclpy.shutdown()


if __name__ == "__main__":
    main()
💡 A shorter angle wrap

atan2(sin(d), cos(d)) gives the same result as the if diff > π … block from Lecture 7, in one line.

Challenge 8.2 · Think · 10 min

Would it work on a car?

Close Gazebo and start the Ackermann LIMO: ros2 launch limo_gazebosim limo_gazebo_ackerman_drive.launch.py. Run go_to_goal to a goal at (0.0, 0.8), which is to the robot's left. What happens, and why? Which line of the controller assumes the robot can turn on the spot?

Discussion
An Ackermann (car-like) robot cannot rotate with linear.x = 0. The line if abs(diff) < 0.4: sets linear to 0 while turning, so the car never moves. Car-like robots need controllers that always move forward while steering. This is why the drive mode matters when you choose a controller, and why Nav2 (Lecture 13) has different controller plugins for them.

What is the difference between open-loop and closed-loop control?

Answer
Open loop sends commands without measuring the result (timing only). Closed loop measures the result (odometry) and corrects the command continuously.

Where in an Odometry message is the robot's y position?

Answer
msg.pose.pose.position.y

The quaternion is (0, 0, 0.707, 0.707). What is the yaw?

Answer
atan2(2·0.707·0.707, 1 − 2·0.5) = atan2(1, 0) = 90° (facing left).

Why must we publish /cmd_vel at least every 0.5 s?

Answer
The twist watchdog stops the robot for safety if commands stop arriving, for example if your node crashes.
