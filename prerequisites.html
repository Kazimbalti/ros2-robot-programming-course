📄 ROS 2 Humble Cheat Sheet

Humble · Ubuntu 22.04 · Python 3.10

1 · Setup

sudo apt install ros-humble-desktop
source /opt/ros/humble/setup.bash
echo "source /opt/ros/humble/setup.bash" >> ~/.bashrc
export ROS_DOMAIN_ID=7        # 0–101, one per team
export ROS_LOCALHOST_ONLY=1   # stay on this PC
printenv | grep ROS           # check
ros2 doctor                   # health check

2 · Workspace

mkdir -p ~/ros2_ws/src && cd ~/ros2_ws
git clone <url> -b humble src/<repo>
sudo rosdep init   # once per PC
rosdep update
rosdep install --from-paths src --ignore-src -r -y
colcon build --symlink-install
source install/setup.bash    # every terminal!

3 · Build options

colcon build --packages-select my_pkg
colcon build --packages-up-to my_pkg
colcon build --parallel-workers 1  # small VM / Pi
colcon list
colcon test && colcon test-result --verbose
rm -rf build install log           # clean build

4 · Create a package

cd ~/ros2_ws/src
# Python
ros2 pkg create my_pkg --build-type ament_python \
  --dependencies rclpy std_msgs --node-name my_node
# C++
ros2 pkg create my_cpp --build-type ament_cmake \
  --dependencies rclcpp std_msgs
ros2 pkg list | grep my_
ros2 pkg executables my_pkg

5 · Nodes

ros2 run <pkg> <exe>
ros2 node list
ros2 node info /turtlesim
# --ros-args options
  -r old_topic:=new_topic    # remap
  -r __ns:=/robot1           # namespace
  -r __node:=new_name        # rename
  -p speed:=0.5              # parameter
  --log-level debug

6 · Topics

ros2 topic list -t
ros2 topic info /scan -v     # + QoS
ros2 topic echo /odom
ros2 topic echo /odom --field pose.pose.position
ros2 topic hz /scan
ros2 topic bw /camera/image_raw
ros2 topic pub -r 10 /cmd_vel geometry_msgs/msg/Twist \
  "{linear: {x: 0.2}, angular: {z: 0.5}}"
ros2 topic pub --once /topic std_msgs/msg/String "{data: hi}"

7 · Services

ros2 service list -t
ros2 service type /spawn
ros2 service call /spawn turtlesim/srv/Spawn \
  "{x: 2.0, y: 2.0, theta: 0.0, name: 'turtle2'}"
ros2 service call /reset std_srvs/srv/Empty

8 · Actions

ros2 action list -t
ros2 action info /turtle1/rotate_absolute
ros2 action send_goal --feedback \
  /turtle1/rotate_absolute \
  turtlesim/action/RotateAbsolute "{theta: 1.57}"

9 · Parameters

ros2 param list /turtlesim
ros2 param get /turtlesim background_r
ros2 param set /turtlesim background_r 150
ros2 param dump /turtlesim > params.yaml
ros2 param load /turtlesim params.yaml
ros2 run pkg exe --ros-args --params-file params.yaml

10 · Interfaces

16 · C++ CMakeLists.txt

cmake_minimum_required(VERSION 3.8)
project(my_cpp)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
add_executable(my_node src/my_node.cpp)
ament_target_dependencies(my_node rclcpp)
install(TARGETS my_node
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config
  DESTINATION share/${PROJECT_NAME})
ament_package()

17 · Minimal launch file

from launch import LaunchDescription
from launch_ros.actions import Node

def generate_launch_description():
    return LaunchDescription([
        Node(package='turtlesim',
             executable='turtlesim_node'),
    ])

18 · Fix it fast

Package not found   → source install/setup.bash
No executable found → entry_points / install()
Echo shows nothing  → ros2 topic info -v (QoS)
Can't see nodes     → same ROS_DOMAIN_ID?
Build freezes       → --parallel-workers 1
/dev/ttyUSB0 denied → usermod -aG dialout $USER
Gazebo won't start  → pkill -f gzserver

ROS 2 Humble Hawksbill (LTS, supported until May 2027). Using Ubuntu 24.04? Replace humble with jazzy. There, ROS_LOCALHOST_ONLY becomes ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST.
