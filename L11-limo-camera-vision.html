11.1The idea in 10 minutes

🖼️

An image is a grid of numbers. A 640 × 480 colour image is 640 × 480 × 3 numbers (Blue, Green, Red: OpenCV uses BGR order). A ROS Image message carries exactly these bytes, plus width, height and an encoding such as rgb8.

🎨

HSV separates "which colour" from "how bright". Hue = the colour (0–180 in OpenCV; red sits at both ends), Saturation = how strong, Value = how bright. A shadow changes V a lot but H very little, so thresholds on H are robust.

flowchart LR CAM["Gazebo camera"] -- "/limo_camera/image
sensor_msgs/Image" --> CT(["color_tracker"]) DEP["depth camera"] -- "/limo_camera/depth/camera_image_depth" --> CT CT -- "/red_target

🎮 GamePut the vision pipeline in order
  1. Subscribe to /limo_camera/image
  2. frame = bridge.imgmsg_to_cv2(msg, "bgr8")
  3. hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
  4. mask = cv2.inRange(hsv, low, high)
  5. contours = cv2.findContours(mask, …) → keep the biggest
  6. Centre x → error in [−1, +1]
  7. Publish /cmd_vel with angular.z = −k × error

L2Lab 2 — cv_bridge + OpenCV basics

LAB 2 · 25 min

image_viewer.py: grey, blur, edges

1Write the node
📄 limo_lab/limo_lab/image_viewer.py
#!/usr/bin/env python3
import cv2
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from cv_bridge import CvBridge


class ImageViewer(Node):
    def __init__(self):
        super().__init__("image_viewer")
        self.bridge_ = CvBridge()
        self.create_subscription(Image, "/limo_camera/image", self.callback_image, 10)
        self.debug_pub_ = self.create_publisher(Image, "/image_viewer/edges", 10)

    def callback_image(self, msg):
        frame = self.bridge_.imgmsg_to_cv2(msg, desired_encoding="bgr8")   # ROS → OpenCV (NumPy)
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)                      # 3 channels → 1
        blur = cv2.GaussianBlur(gray, (5, 5), 0)                            # remove noise
        edges = cv2.Canny(blur, 50, 150)                                    # outlines

        self.get_logger().info(f"image {frame.shape[1]}x{frame.shape[0]}, mean brightness {gray.mean():.0f}",
                               throttle_duration_sec=2.0)
        # publish the result back into ROS so rqt_image_view / RViz can show it
        self.debug_pub_.publish(self.bridge_.cv2_to_imgmsg(edges, encoding="mono8"))

        cv2.imshow("camera", frame)
        cv2.imshow("edges", edges)
        cv2.waitKey(1)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(ImageViewer())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

2Write color_tracker.py (camera + depth)
📄 limo_lab/limo_lab/color_tracker.py
#!/usr/bin/env python3
import cv2
import numpy as np
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from geometry_msgs.msg import Point
from std_msgs.msg import Float32
from cv_bridge import CvBridge


class ColorTracker(Node):
    def __init__(self):
        super().__init__("color_tracker")
        self.declare_parameter("show_window", True)
        self.show_ = self.get_parameter("show_window").value
        self.bridge_ = CvBridge()
        self.depth_ = None                                          # latest depth image (metres)
        self.create_subscription(Image, "/limo_camera/image", self.callback_image, 10)
        self.create_subscription(Image, "/limo_camera/depth/camera_image_depth", self.callback_depth, 10)
        # x = horizontal error (-1 left ... +1 right), z = blob area in pixels (0 = not found)
        self.target_pub_ = self.create_publisher(Point, "red_target", 10)
        self.distance_pub_ = self.create_publisher(Float32, "red_distance", 10)

    def callback_depth(self, msg):
        self.depth_ = self.bridge_.imgmsg_to_cv2(msg, desired_encoding="passthrough")

    def callback_image(self, msg):
        frame = self.bridge_.imgmsg_to_cv2(msg, desired_encoding="bgr8")
        hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
        # red sits at both ends of the hue circle, so we need two ranges
        mask = cv2.inRange(hsv, (0, 120, 70), (10, 255, 255)) | \
               cv2.inRange(hsv, (170, 120, 70), (180, 255, 255))

        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        target = Point()
        if contours:
            biggest = max(contours, key=cv2.contourArea)
            area = cv2.contourArea(biggest)
            if area > 200:                                          # ignore specks of noise
                x, y, w, h = cv2.boundingRect(biggest)
                cx, cy = x + w // 2, y + h // 2
                half = frame.shape[1] / 2
                target.x = (cx - half) / half                       # -1 ... +1
                target.z = float(area)
                cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)
                self.publish_distance(cx, cy)
        self.target_pub_.publish(target)

        if self.show_:
            cv2.imshow("camera", frame)
            cv2.imshow("red mask", mask)
            cv2.waitKey(1)

    def publish_distance(self, cx, cy):
        if self.depth_ is None:
            return
        patch = self.depth_[max(0, cy - 3):cy + 4, max(0, cx - 3):cx + 4]   # 7x7 pixels around the centre
        valid = patch[np.isfinite(patch)]
        if valid.size:
            d = float(np.median(valid))
            self.distance_pub_.publish(Float32(data=d))
            self.get_logger().info(f"red object at {d:.2f} m", throttle_duration_sec=0.5)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(ColorTracker())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

The colour and depth images come from the same simulated camera and have the same size, so the same pixel (cx, cy) gives the object's distance.

11.2Your turn

Activity 11.1 · Core · 30 min

Follow the red box, and stop in front of it

Write red_follower.py. It subscribes to /red_target and /red_distance and publishes /cmd_vel at 10 Hz:

  • Nothing found (z == 0) → turn slowly on the spot to search (angular.z = 0.3).
  • Found → turn toward it: angular.z = −1.0 × x (object on the right means turn right).
  • Drive forward at 0.1 m/s only while the distance is above 0.4 m; then stop.

Test it by dragging the box around in Gazebo. The LIMO should follow it like a puppy.

Hint
Same structure as the wanderer: store the latest Point and distance in callbacks, decide in a 0.1 s timer. Start with self.distance_ = 99.0 so it drives before the first distance arrives.
Solution
📄 limo_lab/limo_lab/red_follower.py
#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Point, Twist
from std_msgs.msg import Float32


class RedFollower(Node):
    def __init__(self):
        super().__init__("red_follower")
        self.declare_parameter("stop_distance", 0.4)
        self.stop_ = self.get_parameter("stop_distance").value
        self.target_ = None
        self.distance_ = 99.0
        self.cmd_pub_ = self.create_publisher(Twist, "cmd_vel", 10)
        self.create_subscription(Point, "red_target", self.callback_target, 10)
        self.create_subscription(Float32, "red_distance", self.callback_distance, 10)
        self.create_timer(0.1, self.control_loop)

    def callback_target(self, msg):
        self.target_ = msg

    def callback_distance(self, msg):
        self.distance_ = msg.data

    def control_loop(self):
        cmd = Twist()
        if self.target_ is None or self.target_.z == 0.0:
            cmd.angular.z = 0.3                          # search
        else:
            cmd.angular.z = -1.0 * self.target_.x        # centre the object
            if self.distance_ > self.stop_:
                cmd.linear.x = 0.1                       # approach
        self.cmd_pub_.publish(cmd)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(RedFollower())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

✅ Knowledge checkLecture 11 check: 5 questions

Why threshold in HSV rather than BGR?

Shadows change V (brightness) a lot and H (colour) very little.

Why does red need two inRange calls?

OpenCV hue runs 0–180 around a circle; red is at both ends.

What does bridge.imgmsg_to_cv2(msg, "bgr8") return?

cv_bridge turns the message bytes into a NumPy array OpenCV can use.

The object's centre is at x = 480 in a 640-wide image. What is the error?

(480 − 320) / 320 = +0.5, so it is on the right and the robot should turn right (negative angular.z).

Your OpenCV window is grey and frozen. What is missing?

imshow only draws when waitKey gives the GUI a moment.
