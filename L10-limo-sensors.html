10.1The idea in 10 minutes

📡

LiDAR = a spinning tape measure. It fires a laser in many directions and measures the distance in each. One sweep is one LaserScan: a list of distances, ranges[], and the angle of the first one plus the step between them.

The LIMO (sim): 360 rays from −2.0 to +2.0 rad (±115°), 0.12–8 m, 8 Hz. Nothing hit gives inf.

📷

Camera = a grid of coloured numbers. A 640-pixel-wide image is a big array. To find a red object: switch to HSV colour (hue is the colour itself), keep only "red" pixels (a mask), find the biggest blob, and take its centre.

L2Lab 2 — Nearest obstacle

LAB 2 · 20 min

nearest_obstacle.py

1Write the node
📄 limo_lab/limo_lab/nearest_obstacle.py
#!/usr/bin/env python3
import math
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan


class NearestObstacle(Node):
    def __init__(self):
        super().__init__("nearest_obstacle")
        self.create_subscription(LaserScan, "scan", self.callback_scan, 10)

    def callback_scan(self, msg):
        nearest_range = math.inf
        nearest_index = -1
        for i, r in enumerate(msg.ranges):
            if math.isfinite(r) and msg.range_min <= r <= msg.range_max and r < nearest_range:
                nearest_range = r
                nearest_index = i

        if nearest_index == -1:
            self.get_logger().info("Nothing in range")
            return
        angle = msg.angle_min + nearest_index * msg.angle_increment
        side = "LEFT" if angle > 0.1 else "RIGHT" if angle < -0.1 else "AHEAD"
        self.get_logger().info(
            f"nearest: {nearest_range:.2f} m at {math.degrees(angle):.0f} deg ({side})",
            throttle_duration_sec=0.5)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(NearestObstacle())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

throttle_duration_sec=0.5 prints at most twice per second, so the terminal stays readable.

L3Lab 3 — The obstacle-avoiding wanderer

LAB 3 · 30 min

Drive forward; if something is close in front, turn toward the more open side

1Write wanderer.py
📄 limo_lab/limo_lab/wanderer.py
#!/usr/bin/env python3
import math
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan
from geometry_msgs.msg import Twist


def sector_min(msg, from_angle, to_angle):
    """Smallest valid range between two angles (radians)."""
    best = math.inf
    for i, r in enumerate(msg.ranges):
        angle = msg.angle_min + i * msg.angle_increment
        if from_angle <= angle <= to_angle and math.isfinite(r) and r >= msg.range_min:
            best = min(best, r)
    return best


class Wanderer(Node):
    def __init__(self):
        super().__init__("wanderer")
        self.declare_parameter("safety_distance", 0.4)   # metres
        self.declare_parameter("speed", 0.15)            # m/s
        self.safety_ = self.get_parameter("safety_distance").value
        self.speed_ = self.get_parameter("speed").value
        self.scan_ = None
        self.cmd_pub_ = self.create_publisher(Twist, "cmd_vel", 10)
        self.create_subscription(LaserScan, "scan", self.callback_scan, 10)
        self.create_timer(0.1, self.control_loop)           # 10 Hz, keeps the watchdog happy

    def callback_scan(self, msg):
        self.scan_ = msg                                     # store...

    def control_loop(self):                                  # ...and decide here
        msg = Twist()
        if self.scan_ is None:
            self.cmd_pub_.publish(msg)
            return
        front = sector_min(self.scan_, -0.35, 0.35)          # about ±20°
        left = sector_min(self.scan_, 0.35, 1.5)
        right = sector_min(self.scan_, -1.5, -0.35)

        if front > self.safety_:
            msg.linear.x = self.speed_                       # path clear: go
        else:
            msg.angular.z = 0.6 if left > right else -0.6    # blocked: turn to the open side
            self.get_logger().info(f"Obstacle {front:.2f} m ahead, turning "
                                   f"{'left' if left > right else 'right'}",
                                   throttle_duration_sec=1.0)
        self.cmd_pub_.publish(msg)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(Wanderer())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

2Write color_tracker.py
📄 limo_lab/limo_lab/color_tracker.py
#!/usr/bin/env python3
import cv2
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from geometry_msgs.msg import Point
from cv_bridge import CvBridge


class ColorTracker(Node):
    def __init__(self):
        super().__init__("color_tracker")
        self.declare_parameter("show_window", True)
        self.show_ = self.get_parameter("show_window").value
        self.bridge_ = CvBridge()
        self.create_subscription(Image, "/limo_camera/image", self.callback_image, 10)
        # x = horizontal error (-1 left ... +1 right), z = blob area in pixels (0 = not found)
        self.target_pub_ = self.create_publisher(Point, "red_target", 10)

    def callback_image(self, msg):
        frame = self.bridge_.imgmsg_to_cv2(msg, desired_encoding="bgr8")   # ROS -> OpenCV
        hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
        # red sits at both ends of the hue circle, so we need two ranges
        mask = cv2.inRange(hsv, (0, 120, 70), (10, 255, 255)) | \
               cv2.inRange(hsv, (170, 120, 70), (180, 255, 255))

        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        target = Point()
        if contours:
            biggest = max(contours, key=cv2.contourArea)
            area = cv2.contourArea(biggest)
            if area > 200:                                      # ignore specks of noise
                x, y, w, h = cv2.boundingRect(biggest)
                cx = x + w / 2
                half = frame.shape[1] / 2
                target.x = (cx - half) / half                   # -1 ... +1
                target.z = float(area)
                cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)
                self.get_logger().info(f"red object: error={target.x:+.2f} area={area:.0f}",
                                       throttle_duration_sec=0.5)
        self.target_pub_.publish(target)

        if self.show_:
            cv2.imshow("camera", frame)
            cv2.imshow("red mask", mask)
            cv2.waitKey(1)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(ColorTracker())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

10.2Your turn

Activity 10.1 · Core · 30 min

Follow the red box

Write red_follower.py. It subscribes to /red_target (from your tracker) and publishes /cmd_vel at 10 Hz:

  • Nothing found (z == 0) → turn slowly on the spot to search (angular.z = 0.3).
  • Found → turn toward it: angular.z = −1.0 × x (object on the right means turn right).
  • Drive forward at 0.1 m/s only if the area is below 15000 (not too close); otherwise stop.

Test it by dragging the box around in Gazebo. The LIMO should follow it like a puppy.

Hint
It's the same structure as the wanderer: store the latest Point in the callback, decide in a 0.1 s timer.
Solution
📄 limo_lab/limo_lab/red_follower.py
#!/usr/bin/env python3
import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Point, Twist


class RedFollower(Node):
    def __init__(self):
        super().__init__("red_follower")
        self.target_ = None
        self.cmd_pub_ = self.create_publisher(Twist, "cmd_vel", 10)
        self.create_subscription(Point, "red_target", self.callback_target, 10)
        self.create_timer(0.1, self.control_loop)

    def callback_target(self, msg):
        self.target_ = msg

    def control_loop(self):
        cmd = Twist()
        if self.target_ is None or self.target_.z == 0.0:
            cmd.angular.z = 0.3                          # search
        else:
            cmd.angular.z = -1.0 * self.target_.x        # centre the object
            if self.target_.z < 15000.0:
                cmd.linear.x = 0.1                       # approach
        self.cmd_pub_.publish(cmd)


def main(args=None):
    rclpy.init(args=args)
    rclpy.spin(RedFollower())
    rclpy.shutdown()


if __name__ == "__main__":
    main()

The scan has angle_min = −2.0 and increment = 0.0111. Which index is straight ahead?

Answer
(0 − (−2.0)) / 0.0111 ≈ index 180.

Why do we convert to HSV before thresholding a colour?

Answer
In HSV the colour (hue) is separate from brightness, so "red" stays red in shadow and light. In BGR, all three numbers change with lighting.

Is the wanderer open-loop or closed-loop? Is it navigation?

Answer
Closed-loop (it reacts to the LiDAR), but reactive only: no map, no goal, so it's not navigation.
